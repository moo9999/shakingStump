-- Roblox Bee Swarm Simulator Improved Teleport Script
local Players = game:GetService("Players")
local TweenService = game:GetService("TweenService")
local LocalPlayer = Players.LocalPlayer
local Character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
local HumanoidRootPart = Character:WaitForChild("HumanoidRootPart")

-- ==================== CONFIGURATION ====================
local TARGET_TOKEN_DECAL_ID = "bloom" -- Looks for 'bloom' inside the token texture name
local TARGET_FIELD_NAME = "Sunflower Field" -- The field to go to afterward
local SPEED = 150 -- Safe speed to bypass anti-cheat kicks
-- =======================================================

-- Safe Anti-Cheat Bypass Movement Function
local function safeMoveTo(targetPosition)
    if not HumanoidRootPart then return end
    
    local distance = (HumanoidRootPart.Position - targetPosition).Magnitude
    local duration = distance / SPEED
    
    local tweenInfo = TweenInfo.new(duration, Enum.EasingStyle.Linear)
    local tween = TweenService:Create(HumanoidRootPart, tweenInfo, {CFrame = CFrame.new(targetPosition) + Vector3.new(0, 3, 0)})
    
    tween:Play()
    tween.Completed:Wait()
end

-- 1. Scan Collectibles for Bloom Tokens
local function findAndCollectToken()
    local collectibles = game.Workspace:FindFirstChild("Collectibles")
    if not collectibles then 
        warn("Collectibles folder not found!")
        return false 
    end
    
    local targetToken = nil
    
    -- Loop through active tokens on the map
    for _, token in pairs(collectibles:GetChildren()) do
        -- Bee Swarm tokens hide their identity in a Decal inside the part
        local frontDecal = token:FindFirstChild("FrontDecal") or token:FindFirstChildOfClass("Decal")
        
        if frontDecal and frontDecal.Texture then
            -- Check if the texture ID contains "bloom" or match identifiers
            if string.find(string.lower(frontDecal.Texture), string.lower(TARGET_TOKEN_DECAL_ID)) then
                targetToken = token
                break
            end
        end
    end
    
    if targetToken then
        print("Found Bloom Shaker Token! Moving...")
        safeMoveTo(targetToken.Position)
        task.wait(0.3) -- Give the server a moment to register pickup
        return true
    else
        warn("No Bloom Shaker tokens found on the map right now.")
        return false
    end
end

-- 2. Move to Field
local function moveToField()
    local flowerZones = game.Workspace:FindFirstChild("FlowerZones")
    if flowerZones then
        local targetField = flowerZones:FindFirstChild(TARGET_FIELD_NAME)
        if targetField then
            print("Moving to field: " .. TARGET_FIELD_NAME)
            safeMoveTo(targetField.Position)
        else
            warn("Field not found: " .. TARGET_FIELD_NAME)
        end
    end
end

-- Execution
local tokenFound = findAndCollectToken()
task.wait(0.5)
moveToField()
